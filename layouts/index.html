// 指数退避 + 全抖动：比「重试 3 次」温柔得多
func Retry(ctx context.Context, fn func() error) error {
delay := 100 * time.Millisecond
for i := 0; i < 6; i++ {
if err := fn(); err == nil {
return nil
}
jitter := time.Duration(rand.Int63n(int64(delay)))
select {
case <-time.After(jitter):
case <-ctx.Done():
return ctx.Err()
}
delay *= 2
}
return errors.New("重试次数耗尽")
}
}
